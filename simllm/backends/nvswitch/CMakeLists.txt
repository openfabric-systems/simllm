cmake_minimum_required(VERSION 3.20)
project(simllm_nvswitch LANGUAGES CXX)
add_library(simllm_nvswitch SHARED switch.cpp)
target_compile_features(simllm_nvswitch PUBLIC cxx_std_17)
set_target_properties(simllm_nvswitch PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(simllm_nvswitch PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MSVC)
    target_compile_options(simllm_nvswitch PRIVATE /W4 /WX)
endif()
